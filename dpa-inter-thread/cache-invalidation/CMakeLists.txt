cmake_minimum_required(VERSION 3.20)
project(dpa_cache_invalidation LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(DOCA_ROOT "/opt/mellanox/doca" CACHE PATH "DOCA installation")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(dpaBuild)
add_compile_options(-fno-lto)
add_link_options(-fno-lto)
add_executable(dpa_cache_invalidation host/main.c host/runtime.c)
target_include_directories(dpa_cache_invalidation PRIVATE host)
target_compile_definitions(dpa_cache_invalidation PRIVATE DPA_APP_NAME=dpa_cache_invalidation_app)
target_compile_options(dpa_cache_invalidation PRIVATE -O2 -Wall -Wextra -Werror -Wno-deprecated-declarations -DFLEXIO_ALLOW_EXPERIMENTAL_API)
dpa_build(dpa_cache_invalidation dpa_cache_invalidation_app)
